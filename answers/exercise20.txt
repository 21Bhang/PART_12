1. todo-app/docker-compose.yml

services:
  app:
    image: todo-front
    build:
      context: ./todo-frontend
      dockerfile: Dockerfile

  backend:
    image: todo-backend
    build:
      context: ./todo-backend
      dockerfile: Dockerfile
    environment:
      - MONGO_URL=mongodb://the_username:the_password@mongo:27017/the_database
      - REDIS_URL=redis://redis:6379

  mongo:
    image: mongo
    environment:
      MONGO_INITDB_ROOT_USERNAME: root
      MONGO_INITDB_ROOT_PASSWORD: example
      MONGO_INITDB_DATABASE: the_database
    volumes:
      - ./todo-backend/mongo/mongo-init.js:/docker-entrypoint-initdb.d/mongo-init.js
      - mongo_data:/data/db

  redis:
    image: redis
    command: ['redis-server', '--appendonly', 'yes']
    volumes:
      - redis_data:/data

  nginx:
    image: nginx:1.29
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf:ro
    ports:
      - 8080:80
    depends_on:
      - app
      - backend

volumes:
  mongo_data:
  redis_data:

The images are named todo-front and todo-backend as required. Production
Dockerfiles are used, no volumes for source code, no dev servers. Only nginx
publishes a port (8080), everything else stays inside the compose network.

2. todo-app/nginx.conf

events { }

http {
  server {
    listen 80;

    location /api/ {
      proxy_pass http://backend:3000/;
    }

    location / {
      proxy_pass http://app:80;
    }
  }
}

The app service is the production frontend (nginx:alpine serving the built
dist) so it listens on port 80 inside its container.

3. todo-frontend/Dockerfile

ENV VITE_BACKEND_URL changed from http://localhost:3000 (exercise 13, backend
outside the container) to http://localhost:8080/api, since the production app
is used through the reverse proxy at port 8080.
